# Android 使用例

## MainActivity の実装

```kotlin
// アプリ全体で単一インスタンスを共有します
object SdkConfig {
    val sdk = KlleonOndeviceSdk()
}

class MainActivity : ComponentActivity() {
    private val klleonOndeviceSdk = SdkConfig.sdk
    private val resourceManager = ResourceManager()

    private val requiredPermissions = arrayOf(
        Manifest.permission.RECORD_AUDIO
    )

    private val permissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        handlePermissionResults(permissions)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        if (checkPermissions()) {
            initializeApp()
        } else {
            requestPermissions()
        }
    }

    private fun initializeApp() {
        setContent {
            OndeviceSampleTheme {
                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = MaterialTheme.colorScheme.background
                ) {
                    WorkerScreen(klleonOndeviceSdk = klleonOndeviceSdk)
                }
            }
        }

        // リソースの初期化（キャッシュへのコピー + YUV 変換 + メモリ読み込み）
        resourceManager.init(
            context = this,
            mp4FileName = "loop_sample.mp4",
            npzFileName = "loop_sample.npz",
            styleFileName = "loop_sample.style",
            avatarId = "character_001",
            onSuccess = {
                // リソースの準備完了 — play() を呼び出せます
            },
            onError = { error ->
                Log.e("MainActivity", "リソースの準備に失敗しました", error)
            }
        )
    }
}
```

## Compose UI の統合

### WorkerScreen Composable

```kotlin
@Composable
fun WorkerScreen(
    klleonOndeviceSdk: KlleonOndeviceSdk,
    viewModel: WorkerViewModel = viewModel()
) {
    val context = LocalContext.current
    var workerView by remember { mutableStateOf<KlleonSdkView?>(null) }
    var messageData by remember { mutableStateOf("") }

    // SDK の状態を collect します
    val accumulatedMessages by viewModel.accumulatedMessages.collectAsState()

    LaunchedEffect(Unit) {
        viewModel.observeSdkState(klleonOndeviceSdk)
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        // SDK の状態表示
        Card(
            modifier = Modifier
                .align(Alignment.TopStart)
                .padding(8.dp),
            colors = CardDefaults.cardColors(
                containerColor = Color.Transparent
            )
        ) {
            Column(modifier = Modifier.padding(12.dp)) {
                Text(
                    text = accumulatedMessages,
                    style = MaterialTheme.typography.bodySmall,
                    color = Color.Black.copy(alpha = 0.8f)
                )
            }
        }

        // 下部のコンテンツ
        Column(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth(),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            // SDK ビュー
            Card(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(400.dp)
            ) {
                AndroidView(
                    factory = { ctx ->
                        KlleonSdkView(ctx).also { view ->
                            workerView = view
                        }
                    },
                    modifier = Modifier.fillMaxSize()
                )
            }

            // メッセージ入力
            OutlinedTextField(
                value = messageData,
                onValueChange = { messageData = it },
                label = { Text("メッセージ") },
                modifier = Modifier.fillMaxWidth()
            )

            // 制御ボタン
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        workerView?.let { view ->
                            CoroutineScope(Dispatchers.IO).launch {
                                klleonOndeviceSdk.play(
                                    context = context,
                                    klleonSdkViewInstance = view,
                                    sdkKey = "YOUR_SDK_KEY",
                                    avatarId = "character_001",
                                    language = "ko_kr",
                                    // sttMode = SttMode.STT_ONLY,  // v0.8.0+ のデフォルト
                                )
                                // 60 秒間の無操作で自動待機画面
                                klleonOndeviceSdk.setIdleTimeout()
                            }
                        }
                    },
                    modifier = Modifier.weight(1f),
                    enabled = workerView != null
                ) {
                    Text("再生")
                }

                Button(
                    onClick = {
                        CoroutineScope(Dispatchers.Default).launch {
                            klleonOndeviceSdk.finish()
                        }
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("終了")
                }
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        klleonOndeviceSdk.sendMessage(messageData)
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("メッセージ送信")
                }

                Button(
                    onClick = {
                        klleonOndeviceSdk.sendMessageEcho(messageData)
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("エコー")
                }

                Button(
                    onClick = {
                        klleonOndeviceSdk.sendStopSpeech()
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("発話中断")
                }
            }

            // 言語切り替えボタン
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        klleonOndeviceSdk.sendChangeLanguage("en")
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("英語に切り替え")
                }
            }
        }
    }
}
```

> v0.8.0 から `voiceRecognitionMode` のデフォルトが `ALWAYS_ON` のため、`play()` 直後にマイクセッションが自動開始されます。マイクボタンは通常不要です — STT の結果については [ALWAYS_ON STT フロー](#always_on-stt-結果の受信stt_only-フロー) の例を参照してください。DISCRETE モード（`startVoiceRecognition`）が必要な場合は [Legacy ガイド](../guides/guides#6-音声認識の統合) を参照してください。

## ALWAYS_ON STT 結果の受信（STT_ONLY フロー）

v0.8.0 の推奨パターンです。`play()` 後に自動開始されるマイクセッションの STT 結果を `sdkState` で受け取り、アプリで後処理したうえで `sendMessage()` を明示的に呼び出します。

```kotlin
@Composable
fun AlwaysOnSttScreen(klleonOndeviceSdk: KlleonOndeviceSdk) {
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.sdkState.collect { state ->
            val json = JSONObject(state)
            val chatType = json.optString("chat_type")
            val message = json.optString("message")

            when (chatType) {
                "STT_RESULT" -> {
                    if (message.isNotBlank()) {
                        // フィルタリング・検証のうえ明示送信します（不適切語の遮断、コマンドの傍受なども可能）
                        klleonOndeviceSdk.sendMessage(message)
                    }
                }
                "STT_AUTH_ERROR" -> {
                    // v0.7.0+: STT トークン V2 attestation の認証失敗
                    Log.e("Stt", "Auth error: $message")
                }
            }
        }
    }
}
```

## 待機画面（Standby）モード

### ウェイクワードによる待機画面の解除

Wakeword マッチングと組み合わせて、音声コマンドで待機画面を解除する例です。
Wakeword マッチングは `enterStandby()` 時に自動的に有効化され、`exitStandby()` 時に自動的に無効化されます。

```kotlin
// Wakeword の登録 — 登録するだけではマッチングは有効になりません（standby 進入時に自動有効化）
klleonOndeviceSdk.registerWakewords(mapOf(
    "ジャパニーズ" to "ja_jp",
    "イングリッシュ" to "en_us",
    "コリアン" to "ko_kr",
))

// Wakeword 検出時に言語を切り替えて standby を解除します
LaunchedEffect(Unit) {
    klleonOndeviceSdk.wakewordDetected.collect { result ->
        klleonOndeviceSdk.sendChangeLanguage(result.tag)
        if (klleonOndeviceSdk.isStandby()) {
            klleonOndeviceSdk.exitStandby()  // マッチングも自動的に無効化されます
        }
    }
}
```

## ViewModel の実装

### WorkerViewModel

```kotlin
class WorkerViewModel : ViewModel() {

    private val _accumulatedMessages = MutableStateFlow("")
    val accumulatedMessages: StateFlow<String> = _accumulatedMessages.asStateFlow()

    fun observeSdkState(klleonOndeviceSdk: KlleonOndeviceSdk) {
        viewModelScope.launch {
            klleonOndeviceSdk.sdkState.collect { state ->
                val messageToAdd = try {
                    val json = JSONObject(state)
                    val chatType = json.optString("chat_type", "")
                    val message = json.getString("message")

                    when (chatType) {
                        "STT_RESULT" -> "[STT] $message"
                        "STT_PARTIAL" -> "[STT 中間] $message"
                        "RESPONSE" -> "[アバター] $message"
                        else -> message
                    }
                } catch (t: Throwable) {
                    state.toString()
                }

                if (messageToAdd.isNotBlank()) {
                    val currentMessages = _accumulatedMessages.value
                    _accumulatedMessages.value = if (currentMessages.isEmpty()) {
                        messageToAdd
                    } else {
                        "$currentMessages\n$messageToAdd"
                    }
                }
            }
        }
    }

    fun clearMessages() {
        _accumulatedMessages.value = ""
    }
}
```

## アバター切り替えの例

`init()` はアバターの変更を自動的に検出します。同じアバターであればバッファを即座に再利用し、別のアバターであれば以前のバッファを解放してから新規に確保します。

```kotlin
class CharacterSelectActivity : ComponentActivity() {

    fun onAvatarSelected(avatarMp4: String, avatarNpz: String, avatarId: String) {
        // init() が avatarId を比較し、バッファの再利用/切り替えを自動処理します
        // 同一アバター: 即座に onSuccess（0ms）
        // 別アバター: 以前のバッファを解放 + 新規確保
        ResourceManager.init(
            context = this,
            mp4FileName = avatarMp4,
            npzFileName = avatarNpz,
            avatarId = avatarId,
            onSuccess = {
                // リソースの準備完了 — ChatActivity へ遷移します
                startActivity(Intent(this, ChatActivity::class.java))
            },
            onError = { error ->
                Log.e("CharacterSelect", "リソースの準備に失敗しました", error)
            }
        )
    }

    override fun onDestroy() {
        super.onDestroy()
        // play() 前に離脱した場合はメモリを解放します
        if (!KlleonOndeviceSdk.playing) {
            ResourceManager.deinit()
        }
    }
}
```

## Wakeword による言語切り替え（Standby 連携）

待機画面（Standby）モードで音声の wakeword を検出し、アバターの会話言語を自動的に切り替えて待機画面を解除する例です。

:::info STT 言語と wakeword 登録の方針
`play()` または `sendChangeLanguage()` に渡した言語が、STT の認識言語としても設定されます。Wakeword は **現在の STT 言語で認識できる単語** として登録する必要があります。

たとえば韓国語モード（`ko_kr`）では `"니혼고"`、`"잉글리쉬"` のような韓国語発音の wakeword が認識され、日本語モード（`ja_jp`）に切り替わると `"韓国語"`、`"英語"` のような日本語の wakeword が認識されます。

したがって、**対応するすべての言語の wakeword をまとめて登録する** ことを推奨します。
:::

:::caution Standby モードが必須です
Wakeword マッチングはデフォルトで無効であり、`enterStandby()` に入ると自動的に有効化されます。`setIdleTimeout()` で自動待機画面を設定するか、手動で `enterStandby()` を呼び出す必要があります。
:::

```kotlin
@Composable
fun WakewordLanguageSwitcher(klleonOndeviceSdk: KlleonOndeviceSdk) {
    var currentLanguage by remember { mutableStateOf("ko_kr") }

    // すべての言語で認識可能な wakeword をまとめて登録します
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.registerWakewords(mapOf(
            // 韓国語 STT で認識
            "니혼고" to "ja_jp", "일본어" to "ja_jp",
            "잉글리쉬" to "en_us", "영어" to "en_us",
            // 日本語 STT で認識
            "韓国語" to "ko_kr", "コリアン" to "ko_kr",
            "英語" to "en_us",
            // 英語 STT で認識
            "Korean" to "ko_kr",
            "Japanese" to "ja_jp",
        ))

        // 自動待機画面を有効化します — standby 進入時に wakeword マッチングも自動的に有効になります
        klleonOndeviceSdk.setIdleTimeout(60_000L)
    }

    // Wakeword 検出時に言語を切り替えて standby を解除します
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.wakewordDetected.collect { result ->
            klleonOndeviceSdk.sendChangeLanguage(result.tag)
            currentLanguage = result.tag
            if (klleonOndeviceSdk.isStandby()) {
                klleonOndeviceSdk.exitStandby()  // マッチングも自動的に無効化されます
            }
        }
    }

    Text(
        text = "現在の言語: $currentLanguage",
        modifier = Modifier.padding(16.dp)
    )
}
```
