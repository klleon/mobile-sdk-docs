# Android Usage Examples

## MainActivity Implementation

```kotlin
// Share a single instance across the app
object SdkConfig {
    val sdk = KlleonOndeviceSdk()
}

class MainActivity : ComponentActivity() {
    private val klleonOndeviceSdk = SdkConfig.sdk
    private val resourceManager = ResourceManager()

    private val requiredPermissions = arrayOf(
        Manifest.permission.RECORD_AUDIO
    )

    private val permissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        handlePermissionResults(permissions)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        if (checkPermissions()) {
            initializeApp()
        } else {
            requestPermissions()
        }
    }

    private fun initializeApp() {
        setContent {
            OndeviceSampleTheme {
                Surface(
                    modifier = Modifier.fillMaxSize(),
                    color = MaterialTheme.colorScheme.background
                ) {
                    WorkerScreen(klleonOndeviceSdk = klleonOndeviceSdk)
                }
            }
        }

        // Initialize resources (copy to cache + YUV conversion + load into memory)
        resourceManager.init(
            context = this,
            mp4FileName = "loop_sample.mp4",
            npzFileName = "loop_sample.npz",
            styleFileName = "loop_sample.style",
            avatarId = "character_001",
            onSuccess = {
                // Resources ready — play() can now be called
            },
            onError = { error ->
                Log.e("MainActivity", "Failed to prepare resources", error)
            }
        )
    }
}
```

## Compose UI Integration

### WorkerScreen Composable

```kotlin
@Composable
fun WorkerScreen(
    klleonOndeviceSdk: KlleonOndeviceSdk,
    viewModel: WorkerViewModel = viewModel()
) {
    val context = LocalContext.current
    var workerView by remember { mutableStateOf<KlleonSdkView?>(null) }
    var messageData by remember { mutableStateOf("") }

    // Collect SDK state
    val accumulatedMessages by viewModel.accumulatedMessages.collectAsState()

    LaunchedEffect(Unit) {
        viewModel.observeSdkState(klleonOndeviceSdk)
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        // Display SDK state
        Card(
            modifier = Modifier
                .align(Alignment.TopStart)
                .padding(8.dp),
            colors = CardDefaults.cardColors(
                containerColor = Color.Transparent
            )
        ) {
            Column(modifier = Modifier.padding(12.dp)) {
                Text(
                    text = accumulatedMessages,
                    style = MaterialTheme.typography.bodySmall,
                    color = Color.Black.copy(alpha = 0.8f)
                )
            }
        }

        // Bottom content
        Column(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth(),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            // SDK view
            Card(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(400.dp)
            ) {
                AndroidView(
                    factory = { ctx ->
                        KlleonSdkView(ctx).also { view ->
                            workerView = view
                        }
                    },
                    modifier = Modifier.fillMaxSize()
                )
            }

            // Message input
            OutlinedTextField(
                value = messageData,
                onValueChange = { messageData = it },
                label = { Text("Message") },
                modifier = Modifier.fillMaxWidth()
            )

            // Control buttons
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        workerView?.let { view ->
                            CoroutineScope(Dispatchers.IO).launch {
                                klleonOndeviceSdk.play(
                                    context = context,
                                    klleonSdkViewInstance = view,
                                    sdkKey = "YOUR_SDK_KEY",
                                    avatarId = "character_001",
                                    language = "ko_kr",
                                    // sttMode = SttMode.STT_ONLY,  // v0.8.0+ default
                                )
                                // Automatic standby after 60 seconds of inactivity
                                klleonOndeviceSdk.setIdleTimeout()
                            }
                        }
                    },
                    modifier = Modifier.weight(1f),
                    enabled = workerView != null
                ) {
                    Text("Play")
                }

                Button(
                    onClick = {
                        CoroutineScope(Dispatchers.Default).launch {
                            klleonOndeviceSdk.finish()
                        }
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Finish")
                }
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        klleonOndeviceSdk.sendMessage(messageData)
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Send message")
                }

                Button(
                    onClick = {
                        klleonOndeviceSdk.sendMessageEcho(messageData)
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Echo")
                }

                Button(
                    onClick = {
                        klleonOndeviceSdk.sendStopSpeech()
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Stop speech")
                }
            }

            // Language switch button
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Button(
                    onClick = {
                        klleonOndeviceSdk.sendChangeLanguage("en")
                    },
                    modifier = Modifier.weight(1f)
                ) {
                    Text("Switch to English")
                }
            }
        }
    }
}

A microphone button is generally unnecessary — see the [ALWAYS_ON STT flow](#always_on-stt-result-reception-stt_only-flow) example for STT results. If you need DISCRETE mode (`startVoiceRecognition`), see the [legacy guide](../guides/guides#6-speech-recognition-integration).

## ALWAYS_ON STT Result Reception (STT_ONLY Flow)

The recommended pattern for v0.8.0. Receive STT results from the microphone session that starts automatically after `play()` through `sdkState`, and call `sendMessage()` explicitly after your app post-processes them.

```kotlin
@Composable
fun AlwaysOnSttScreen(klleonOndeviceSdk: KlleonOndeviceSdk) {
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.sdkState.collect { state ->
            val json = JSONObject(state)
            val chatType = json.optString("chat_type")
            val message = json.optString("message")

            when (chatType) {
                "STT_RESULT" -> {
                    if (message.isNotBlank()) {
                        // Send explicitly after filtering/validation (profanity blocking, command interception, etc.)
                        klleonOndeviceSdk.sendMessage(message)
                    }
                }
                "STT_AUTH_ERROR" -> {
                    // v0.7.0+: STT token V2 attestation authentication failure
                    Log.e("Stt", "Auth error: $message")
                }
            }
        }
    }
}
```

## Standby Mode

### Exiting Standby with a Wake Word

An example of exiting standby by voice command in combination with wakeword matching.
Wakeword matching is enabled automatically on `enterStandby()` and disabled automatically on `exitStandby()`.

```kotlin
// Register wakewords — registration alone does not enable matching (enabled automatically when standby is entered)
klleonOndeviceSdk.registerWakewords(mapOf(
    "ジャパニーズ" to "ja_jp",
    "イングリッシュ" to "en_us",
    "コリアン" to "ko_kr",
))

// Switch language and exit standby when a wakeword is detected
LaunchedEffect(Unit) {
    klleonOndeviceSdk.wakewordDetected.collect { result ->
        klleonOndeviceSdk.sendChangeLanguage(result.tag)
        if (klleonOndeviceSdk.isStandby()) {
            klleonOndeviceSdk.exitStandby()  // Matching is disabled automatically as well
        }
    }
}
```

## ViewModel Implementation

### WorkerViewModel

```kotlin
class WorkerViewModel : ViewModel() {

    private val _accumulatedMessages = MutableStateFlow("")
    val accumulatedMessages: StateFlow<String> = _accumulatedMessages.asStateFlow()

    fun observeSdkState(klleonOndeviceSdk: KlleonOndeviceSdk) {
        viewModelScope.launch {
            klleonOndeviceSdk.sdkState.collect { state ->
                val messageToAdd = try {
                    val json = JSONObject(state)
                    val chatType = json.optString("chat_type", "")
                    val message = json.getString("message")

                    when (chatType) {
                        "STT_RESULT" -> "[STT] $message"
                        "STT_PARTIAL" -> "[STT interim] $message"
                        "RESPONSE" -> "[Avatar] $message"
                        else -> message
                    }
                } catch (t: Throwable) {
                    state.toString()
                }

                if (messageToAdd.isNotBlank()) {
                    val currentMessages = _accumulatedMessages.value
                    _accumulatedMessages.value = if (currentMessages.isEmpty()) {
                        messageToAdd
                    } else {
                        "$currentMessages\n$messageToAdd"
                    }
                }
            }
        }
    }

    fun clearMessages() {
        _accumulatedMessages.value = ""
    }
}
```

## Avatar Switching Example

`init()` automatically detects avatar changes. For the same avatar it reuses the buffers immediately; for a different avatar it releases the previous buffers and allocates new ones.

```kotlin
class CharacterSelectActivity : ComponentActivity() {

    fun onAvatarSelected(avatarMp4: String, avatarNpz: String, avatarId: String) {
        // init() compares avatarId and handles buffer reuse/switching automatically
        // Same avatar: immediate onSuccess (0 ms)
        // Different avatar: release previous buffers + new allocation
        ResourceManager.init(
            context = this,
            mp4FileName = avatarMp4,
            npzFileName = avatarNpz,
            avatarId = avatarId,
            onSuccess = {
                // Resources ready — move to ChatActivity
                startActivity(Intent(this, ChatActivity::class.java))
            },
            onError = { error ->
                Log.e("CharacterSelect", "Failed to prepare resources", error)
            }
        )
    }

    override fun onDestroy() {
        super.onDestroy()
        // Release memory if leaving before play()
        if (!KlleonOndeviceSdk.playing) {
            ResourceManager.deinit()
        }
    }
}
```

@Composable
fun SDKStateObserver(klleonOndeviceSdk: KlleonOndeviceSdk) {
    var currentState by remember { mutableStateOf("") }

    LaunchedEffect(klleonOndeviceSdk) {
        klleonOndeviceSdk.sdkState.collect { state ->

        text = currentState,
        modifier = Modifier.padding(16.dp)
    )
}

## Wakeword-based Language Switching (Standby Integration)

An example that detects a voice wakeword in standby mode to automatically switch the avatar's conversation language and exit standby.

:::info STT language and wakeword registration strategy
The language you pass to `play()` or `sendChangeLanguage()` is also set as the STT recognition language. Wakewords must be registered as **words the current STT language can recognize**.

For example, in Korean mode (`ko_kr`) wakewords with Korean pronunciation such as `"니혼고"` and `"잉글리쉬"` are recognized, and after switching to Japanese mode (`ja_jp`) Japanese wakewords such as `"韓国語"` and `"英語"` are recognized.

Therefore, it is best to **register the wakewords for all supported languages at once**.
:::

:::caution Standby mode required
Wakeword matching is disabled by default and is enabled automatically when `enterStandby()` is entered. You must configure automatic standby with `setIdleTimeout()` or call `enterStandby()` manually for wakeword matching to work.
:::

```kotlin
@Composable
fun WakewordLanguageSwitcher(klleonOndeviceSdk: KlleonOndeviceSdk) {
    var currentLanguage by remember { mutableStateOf("ko_kr") }

    // Register wakewords recognizable in every language at once
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.registerWakewords(mapOf(
            // Recognized by Korean STT
            "니혼고" to "ja_jp", "일본어" to "ja_jp",
            "잉글리쉬" to "en_us", "영어" to "en_us",
            // Recognized by Japanese STT
            "韓国語" to "ko_kr", "コリアン" to "ko_kr",
            "英語" to "en_us",
            // Recognized by English STT
            "Korean" to "ko_kr",
            "Japanese" to "ja_jp",
        ))

        // Enable automatic standby — wakeword matching is also enabled automatically when standby is entered
        klleonOndeviceSdk.setIdleTimeout(60_000L)
    }

    // Switch language and exit standby when a wakeword is detected
    LaunchedEffect(Unit) {
        klleonOndeviceSdk.wakewordDetected.collect { result ->
            klleonOndeviceSdk.sendChangeLanguage(result.tag)
            currentLanguage = result.tag
            if (klleonOndeviceSdk.isStandby()) {
                klleonOndeviceSdk.exitStandby()  // Matching is disabled automatically as well
            }
        }
    }

    Text(
        text = "Current language: $currentLanguage",
        modifier = Modifier.padding(16.dp)
    )
}
```
